# Riptide — dependencies
#
# Everything the workspace needs to build and run, grouped by how you install it.
# Target platform: ROS 2 Jazzy on Ubuntu 24.04.
#
# Fast path (installs everything with a rosdep key below, EXCEPT the two
# manual-install libraries in section 4 — Pinocchio and MuJoCo):
#
#   source /opt/ros/jazzy/setup.bash
#   rosdep install --from-paths . --ignore-src -r -y
#   colcon build --symlink-install
#   source install/setup.bash
#
# ============================================================================
# 1. ROS 2 base
# ============================================================================
# ROS 2 Jazzy (ros-jazzy-desktop recommended — pulls in rviz2, tf2, launch, etc.)
#   sudo apt install ros-jazzy-desktop
ros-jazzy-desktop

# ============================================================================
# 2. ROS 2 packages (apt: ros-jazzy-<name>, or via `rosdep install`)
# ============================================================================

# 3. System libraries (apt / pip; resolved by rosdep keys)
# ============================================================================
eigen               # libeigen3-dev            (riptide_control, riptide_dynamics)
python3-numpy       # python3-numpy            (riptide_eval)
python3-matplotlib  # python3-matplotlib       (riptide_eval — benchmark plots)

# ============================================================================
# 4. External libraries — NOT rosdep/apt keys, install manually
# ============================================================================
# These are picked up by CMake via *_ROOT (env var or -D...). They are the two
# heavy, non-standard dependencies of the project.
#
# --- Pinocchio (riptide_dynamics) — whole-body FK/Jacobian/M/Coriolis ---
#   CMake: PINOCCHIO_ROOT (env or -D), default ~/.local/pinocchio
#   Option A (apt):    sudo apt install ros-jazzy-pinocchio
#                      then export PINOCCHIO_ROOT=/opt/ros/jazzy
#   Option B (source): build with python/collision OFF, urdf ON into
#                      ~/.local/pinocchio
pinocchio           # >= 3.x   (manual: see README "Prerequisites")

# --- MuJoCo SDK (riptide_mujoco) — physics engine (headers + libmujoco) ---
#   CMake: MUJOCO_ROOT (env or -D), default ~/.mujoco/mujoco-3.10.0
#   Download the SDK from https://github.com/google-deepmind/mujoco/releases
#   and unpack it there (or point MUJOCO_ROOT at it).
mujoco==3.10.0      # manual: SDK under ~/.mujoco/mujoco-3.10.0

# ============================================================================
# 5. Test dependencies (only needed to run `colcon test`)
# ============================================================================
ament_lint_auto             # ros-jazzy-ament-lint-auto
ament_lint_common           # ros-jazzy-ament-lint-common
ament_cmake_gtest           # ros-jazzy-ament-cmake-gtest
ament_copyright             # ros-jazzy-ament-copyright
ament_flake8                # ros-jazzy-ament-flake8
ament_pep257                # ros-jazzy-ament-pep257
